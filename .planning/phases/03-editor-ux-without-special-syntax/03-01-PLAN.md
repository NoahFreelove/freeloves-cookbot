    ```javascript
    // recipe-chip-composer.js
    // Phase 3 (D-A1, D-D3, D-D4): caret-anchored MudAutocomplete + cooking-mode scroll-highlight + fail-soft probe.
    // Module shape mirrors cooking-timers.js (window.<Name> = { ... }; no ES modules).
    window.RecipeChipComposer = {
        // D-D4: probe called by C# on first render; if this throws or returns non-"ok",
        // the composer falls back to MudTextField Lines=3 (raw [name](#id) text editing).
        ping() {
            return "ok";
        },

        // D-A1: returns the caret pixel position relative to the parent element rect.
        // Used by C# to absolutely-position the MudAutocomplete<Ingredient> popover at the @-trigger location.
        // For contenteditable spans, uses Selection/Range API (no mirror-div needed).
        getCaretCoords(elementId) {
            const el = document.getElementById(elementId);
            if (!el) return null;
            const rect = el.getBoundingClientRect();
            const sel = window.getSelection();
            if (sel && sel.rangeCount > 0) {
                const r = sel.getRangeAt(0).cloneRange();
                r.collapse(true);
                const cr = r.getClientRects()[0];
                if (cr) {
                    return { x: cr.left - rect.left, y: cr.bottom - rect.top };
                }
            }
            // Fallback: bottom-left of the element.
            return { x: 0, y: rect.height };
        },

        // D-D3: cooking-mode ingredient-chip click scrolls the sidebar item into view and pulses a highlight class.
        // Idempotent across rapid clicks via per-element timer cancellation.
        scrollIntoViewWithHighlight(elementId, highlightClass, durationMs) {
            highlightClass = highlightClass || "chip-highlight-pulse";
            durationMs = durationMs || 1500;
            const el = document.getElementById(elementId);
            if (!el) return false;
            el.scrollIntoView({ behavior: "smooth", block: "center" });
            // Cancel any prior timer to avoid flicker on rapid re-clicks.
            if (el._highlightTimer) {
                clearTimeout(el._highlightTimer);
                el.classList.remove(highlightClass);
            }
            el.classList.add(highlightClass);
            el._highlightTimer = setTimeout(() => {
                el.classList.remove(highlightClass);
                el._highlightTimer = null;
            }, durationMs);
            return true;
        }
    };
    ```

    ```razor
    @using System.Text.RegularExpressions
    @using Microsoft.JSInterop
    @inject IJSRuntime JS

    @if (_jsInteropAvailable && Interactive)

        [Parameter] public bool Interactive { get; set; } = true;
        [Parameter] public string Text { get; set; } = string.Empty;
        [Parameter] public EventCallback<string> TextChanged { get; set; }
        [Parameter] public IReadOnlyList<ParsedIngredient> Ingredients { get; set; } = Array.Empty<ParsedIngredient>();
        [Parameter] public EventCallback<int> OnIngredientChipClick { get; set; }

        private bool _jsInteropAvailable;
        private bool _pickerVisible;
        private double _pickerX, _pickerY;
        private MudBlazor.MudAutocomplete<ParsedIngredient>? _pickerRef;
        private readonly string _composerId = $"composer-{Guid.NewGuid():N}";

        public sealed record StepToken(int StartIndex, int Length, bool IsChip, string Text, string DisplayName = "", int IngredientId = 0);

        protected override async Task OnAfterRenderAsync(bool firstRender)
        {
            if (!firstRender) return;
            try
            {
                var pong = await JS.InvokeAsync<string>("RecipeChipComposer.ping");
                _jsInteropAvailable = pong == "ok";
            }
            catch (JSException) { _jsInteropAvailable = false; }
            catch (JSDisconnectedException) { _jsInteropAvailable = false; }
            catch (TaskCanceledException) { _jsInteropAvailable = false; }
            StateHasChanged();
        }

        // Internal helper exposed for bUnit testing the D-A1 invariant (see RESEARCH.md Pattern 1).
        internal string SimulateInsertChip(ParsedIngredient ing, int caretIndex, int partialAtTokenLength)
        {
            var chipMarkdown = $"[{ing.Name}](#{ing.LocalId})";
            var startReplace = Math.Max(0, caretIndex - partialAtTokenLength);
            Text = Text.Substring(0, startReplace) + chipMarkdown + Text.Substring(caretIndex);
            return Text;
        }

        // Public helpers used by the bUnit test scaffolding.
        internal Task SimulateAtTriggerSelectionAsync(ParsedIngredient ing, int caretIndex, int partialAtTokenLength)
        {
            var newText = SimulateInsertChip(ing, caretIndex, partialAtTokenLength);
            return OnTextChanged(newText);
        }

        internal Task SimulateButtonInsertionAsync(ParsedIngredient ing, int caretIndex)
            => SimulateAtTriggerSelectionAsync(ing, caretIndex, partialAtTokenLength: 0);

        private IEnumerable<StepToken> TokenizeText(string text)
        {
            if (string.IsNullOrEmpty(text))
            {
                yield return new StepToken(0, 0, IsChip: false, Text: "");
                yield break;
            }
            var last = 0;
            foreach (Match m in IngredientLinkPatterns.Pattern.Matches(text))
            {
                if (m.Index > last)
                    yield return new StepToken(last, m.Index - last, IsChip: false, Text: text.Substring(last, m.Index - last));
                var id = int.Parse(m.Groups[2].Value);
                yield return new StepToken(m.Index, m.Length, IsChip: true,
                    Text: m.Value, DisplayName: m.Groups[1].Value, IngredientId: id);
                last = m.Index + m.Length;
            }
            if (last < text.Length)
                yield return new StepToken(last, text.Length - last, IsChip: false, Text: text.Substring(last));
        }

        private bool LookupResolved(int id) => Ingredients.Any(i => i.LocalId == id);

        private async Task OnTextChanged(string newText)
        {
            Text = newText ?? string.Empty;
            if (TextChanged.HasDelegate) await TextChanged.InvokeAsync(Text);
        }

        private async Task RemoveChip(int start, int length)
        {
            await OnTextChanged(Text.Substring(0, start) + Text.Substring(start + length));
        }

        private Task RequestReplace(int start, int length, int currentId)
        {
            // Open picker positioned at chip; selection replaces this chip's range.
            _pickerVisible = true;
            _pendingReplace = (start, length);
            return Task.CompletedTask;
        }

        private (int Start, int Length)? _pendingReplace;

        private async Task OpenInsertPicker()
        {
            _pendingReplace = null;
            try
            {
                var coords = await JS.InvokeAsync<Dictionary<string, double>?>("RecipeChipComposer.getCaretCoords", _composerId);
                if (coords != null) { _pickerX = coords.GetValueOrDefault("x"); _pickerY = coords.GetValueOrDefault("y"); }
            }
            catch (JSDisconnectedException) { /* graceful */ }
            _pickerVisible = true;
        }

        private async Task<IEnumerable<ParsedIngredient>> SearchIngredients(string value, CancellationToken ct)
        {
            if (string.IsNullOrWhiteSpace(value)) return Ingredients.Take(10);
            var v = value.ToLowerInvariant();
            return Ingredients.Where(i => i.Name.ToLowerInvariant().Contains(v)).Take(10);
        }

        private async Task OnIngredientSelected(ParsedIngredient? ing)
        {
            if (ing == null) { _pickerVisible = false; return; }
            if (_pendingReplace is { } range)
            {
                var newText = Text.Substring(0, range.Start)
                              + $"[{ing.Name}](#{ing.LocalId})"
                              + Text.Substring(range.Start + range.Length);
                await OnTextChanged(newText);
                _pendingReplace = null;
            }
            else
            {
                // Append at end as a baseline for "Insert ingredient" button when no caret context.
                var newText = (Text.Length > 0 && !Text.EndsWith(' ') ? Text + " " : Text) + $"[{ing.Name}](#{ing.LocalId})";
                await OnTextChanged(newText);
            }
            _pickerVisible = false;
        }

        private Task OnSegmentKeyDown(Microsoft.AspNetCore.Components.Web.KeyboardEventArgs e, int start, int length)
        {
            // EDITOR-07: Backspace at position 0 of a text segment immediately after a chip removes the chip.
            // (Detection via JS-side caret position is a follow-up refinement; placeholder here.)
            return Task.CompletedTask;
        }

        private Task OnSegmentInput(Microsoft.AspNetCore.Components.ChangeEventArgs e, int start, int length)
        {
            var newSegment = e.Value?.ToString() ?? string.Empty;
            var newText = Text.Substring(0, start) + newSegment + Text.Substring(start + length);
            return OnTextChanged(newText);
        }

        private async Task InvokeIngredientChipClick(int ingredientId)
        {
            if (OnIngredientChipClick.HasDelegate)
            {
                try
                {
                    await JS.InvokeAsync<bool>("RecipeChipComposer.scrollIntoViewWithHighlight", $"ingredient-{ingredientId}");
                }
                catch (JSDisconnectedException) { /* graceful per Pitfall 4 */ }
                await OnIngredientChipClick.InvokeAsync(ingredientId);
            }
        }
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user keyboard/paste → contenteditable text segment | Untrusted text reaches Blazor circuit; if `contenteditable` accepts HTML pasting, attacker-controlled HTML can be rendered. |
| C# circuit → JS interop module | `IJSRuntime.InvokeAsync` arguments cross the trust boundary; module receives `elementId` strings and reads/mutates DOM. |
| JS module DOM → user-visible page | `scrollIntoView` and class-toggle target attacker-influenceable element ids. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03P01-01 | Tampering / Information Disclosure | `<span contenteditable>` text segments accepting pasted HTML (Pitfall 7) | mitigate | Use `contenteditable="plaintext-only"` (D-A4 segments). Browser strips HTML on paste. Action: literal attribute set in the chip-flow render branch; CSS-supported on Chrome/Edge/Safari/Firefox 2023+ (Assumption A3). Older Firefox falls back to full contenteditable — accepted residual risk; the trusted-LAN posture makes the realistic attacker surface small. |
| T-03P01-02 | Information Disclosure | JS module receives `elementId` from C#; if id were attacker-supplied (e.g., from a malicious recipe), it could target arbitrary DOM nodes. | accept | All `elementId`s used in this plan are constructed in C# (`composer-{Guid}` for caret coords; `ingredient-{int}` for cooking-mode scroll). No user-controlled string flows to `getElementById`. Trivially compliant — accepted with no mitigation. |
| T-03P01-03 | Denial of Service | `getCaretCoords` called on every `@`-trigger keystroke could flood the circuit. | accept | The MudAutocomplete debounce + 500ms input debounce (`Immediate=true DebounceInterval=500`) bounds invocation rate. No new DoS surface vs. existing editor. Accept. |
| T-03P01-04 | Elevation of Privilege | `MudMenu` opens via JS interop; if module load fails, menu cannot be opened. | accept | The component falls back to `MudTextField` when JS fails (D-D4); user can still author/save. No privilege uplift possible from a fail-soft fallback. Accept. |
