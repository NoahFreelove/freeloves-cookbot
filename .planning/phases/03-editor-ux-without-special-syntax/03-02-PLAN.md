        public enum StepKind { Step, Section }

        [Parameter] public ParsedStep Step { get; set; } = null!;

        // Resets to false (chip view) on every component re-creation (save+reload, navigation, etc.).
        private bool _showRawMarkdown;

        protected override void OnParametersSet()

            _kind = Step.IsSection ? StepKind.Section : StepKind.Step;

        private async Task OnKindRequested(StepKind requested)
        {
            if (requested == _kind) return;

            // Pitfall 5: do NOT flip _kind until confirmed.
            if (requested == StepKind.Section)
            {
                var timerCount = Step.Timers?.Count ?? 0;
                var refCount = CountIngredientRefs(Step.Text);
                if (timerCount > 0 || refCount > 0)
                {
                    var options = new DialogOptions { MaxWidth = MaxWidth.Small, FullWidth = false, CloseOnEscapeKey = true };

                    {
                        ["TimerCount"] = timerCount,
                        ["RefCount"] = refCount
                    };

                    if (result is null || result.Canceled)
                    {
                        StateHasChanged(); // re-render to revert visual toggle state
                        return;
                    }
                }

            }

            {
                Step.IsSection = false;
                _kind = StepKind.Step;
            }
