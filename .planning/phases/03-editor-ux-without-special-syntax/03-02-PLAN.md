    ```razor
    @using CookBot.Application.Recipes
    @using CookBot.Domain.Entities
    @using CookBot.Domain.Interfaces
    @using System.Text.RegularExpressions

    <MudPaper Class="pa-3 mb-3" Elevation="0" Style="border: 1px solid #e0e0e0; border-radius: 8px;">

                            @foreach (var t in Step.Timers.ToList())
                            {
                                var captured = t;
                                <TimerChip Timer="@captured"
                                           OnRemove="@(() => RemoveTimer(captured))"
                                           OnChanged="@(updated => UpdateTimer(captured, updated))" />

    @code {
        public enum StepKind { Step, Section }

        [Parameter] public ParsedStep Step { get; set; } = null!;
        [Parameter] public int Index { get; set; }
        [Parameter] public bool IsLast { get; set; }
        [Parameter] public IReadOnlyList<ParsedIngredient> Ingredients { get; set; } = Array.Empty<ParsedIngredient>();
        [Parameter] public EventCallback OnRemove { get; set; }
        [Parameter] public EventCallback OnMoveUp { get; set; }
        [Parameter] public EventCallback OnMoveDown { get; set; }

        private StepKind _kind;

        protected override void OnParametersSet()
        {
            _kind = Step.IsSection ? StepKind.Section : StepKind.Step;
        }

        private async Task OnKindRequested(StepKind requested)
        {
            if (requested == _kind) return;

            // Pitfall 5: do NOT flip _kind until confirmed.
            if (requested == StepKind.Section)
            {
                var timerCount = Step.Timers?.Count ?? 0;
                var refCount = CountIngredientRefs(Step.Text);
                if (timerCount > 0 || refCount > 0)
                {
                    var options = new DialogOptions { MaxWidth = MaxWidth.Small, FullWidth = false, CloseOnEscapeKey = true };
                    var parameters = new DialogParameters
                    {
                        ["TimerCount"] = timerCount,
                        ["RefCount"] = refCount
                    };
                    var dialog = await DialogService.ShowAsync<SectionDropConfirmationDialog>("Convert?", parameters, options);
                    var result = await dialog.Result;
                    if (result is null || result.Canceled)
                    {
                        StateHasChanged(); // re-render to revert visual toggle state
                        return;
                    }
                }

                // D-B2: section heading reuses current step text. Drop timers + ingredient refs (Phase 1 D-02 SectionStep has neither).
                Step.IsSection = true;
                Step.Timers = new List<StepTimer>();
                // Strip [name](#id) markdown from heading text — section heading is plain text only.
                Step.Text = IngredientLinkPatterns.Pattern.Replace(Step.Text, m => m.Groups[1].Value);
                _kind = StepKind.Section;
            }
            else // Section → Step
            {
                Step.IsSection = false;
                _kind = StepKind.Step;
            }
            StateHasChanged();
        }

        private static int CountIngredientRefs(string text)
            => string.IsNullOrEmpty(text) ? 0 : IngredientLinkPatterns.Pattern.Matches(text).Count;

        private Task OnTextChanged(string newText)
        {
            Step.Text = newText ?? string.Empty;
            return Task.CompletedTask;
        }

        private Task OnHeadingChanged(string newHeading)
        {
            Step.Text = newHeading ?? string.Empty;
            return Task.CompletedTask;
        }

        private void RemoveTimer(StepTimer t)
        {
            Step.Timers?.Remove(t);
            StateHasChanged();
        }

        private void UpdateTimer(StepTimer original, StepTimer updated)
        {
            // Mutated in place via TimerChip → original IS updated. StateHasChanged for re-render.
            StateHasChanged();
        }
    }
    ```
