    ```csharp
    // Cache key: invalidated whenever CurrentStep changes (step navigation).
    private RecipeStep? _refIdsCacheStep;
    private HashSet<int> _refIdsCache = new();

    private HashSet<int> CurrentStepRefIds()
    {
        // Cheap reference-equality check — Next/Previous step swap CurrentStep to a different RecipeStep instance.
        if (ReferenceEquals(_refIdsCacheStep, CurrentStep)) return _refIdsCache;

        var ids = new HashSet<int>();
        var text = CurrentStep.Text ?? string.Empty;
        foreach (Match m in IngredientLinkPatterns.Pattern.Matches(text))
        {
            if (int.TryParse(m.Groups[2].Value, out var id) &amp;&amp; id > 0)
                ids.Add(id);
        }
        _refIdsCache = ids;
        _refIdsCacheStep = CurrentStep;
        return ids;
    }
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03P06-01 | Denial of Service | IngredientLinkPatterns.Pattern.Matches | accept | The regex `\[([^\]]*)\]\(#(\d+)\)` is linear-time on bounded input. No nested quantifiers. RegexOptions.Compiled enabled. Step text is bounded by Phase 1 validator (typically <500 chars). Same posture WR-04 logged for TimerDetectionService — accepted, not a security concern. |
| T-03P06-02 | Tampering | int.TryParse on Groups[2].Value | mitigate | `TryParse` (not `Parse`) — overflow / non-numeric returns false, the id is skipped, no throw. Additional `id > 0` filter rejects zero/negative ids that would be invalid `RecipeLocalId` values. |
| T-03P06-03 | Information Disclosure | Sidebar highlight reveals which ingredients are in step text | accept | The recipe is already accessible to this user (UserGuard + UserCanAccessRecipeAsync gate at line ~275). Highlight is purely cosmetic over already-authorized data. |
