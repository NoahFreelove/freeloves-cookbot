           public void GetSchema_IsCachedAcrossCalls() {
               var provider = new RecipeJsonSchemaProvider();
               Assert.Same(provider.GetSchema(), provider.GetSchema());

           private static void WalkAndCheck(JsonNode? node, string path, System.Collections.Generic.List<string> bad) {
               if (node is JsonObject obj) {
                   var isObjectSchema = false;
                   if (obj["type"] is JsonValue tv && tv.TryGetValue<string>(out var ts) && ts == "object") isObjectSchema = true;
                   else if (obj["type"] is JsonArray ta && ta.Any(x => x is JsonValue v && v.TryGetValue<string>(out var s) && s == "object")) isObjectSchema = true;
                   else if (obj.ContainsKey("properties")) isObjectSchema = true;
                   if (isObjectSchema && obj["additionalProperties"]?.GetValue<bool>() != false) {
                       bad.Add(path);
                   }
                   foreach (var kvp in obj) WalkAndCheck(kvp.Value, $"{path}.{kvp.Key}", bad);
               } else if (node is JsonArray arr) {
                   for (int i = 0; i < arr.Count; i++) WalkAndCheck(arr[i], $"{path}[{i}]", bad);
               }
           }
