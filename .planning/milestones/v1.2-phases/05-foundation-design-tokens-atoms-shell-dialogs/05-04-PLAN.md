CbDialogService API (the Phase-7 migration target — design for ergonomic 1:1 swap with IDialogService):

```csharp
public interface ICbDialogService
{
    // Generic — equivalent to MudBlazor's DialogService.ShowAsync<T>
    Task<CbDialogResult> ShowAsync<TDialog>(
        string title,
        CbDialogParameters? parameters = null,
        CbDialogOptions? options = null
    ) where TDialog : ComponentBase;

    // Type-erased fallback for dynamic content
    Task<CbDialogResult> ShowAsync(
        Type dialogType,
        string title,
        CbDialogParameters? parameters = null,
        CbDialogOptions? options = null
    );

    // Internal: CbDialogHost subscribes here; sandbox can also subscribe for the demo
    event Func<CbDialogRequest, Task>? OnRequest;
}

public sealed record CbDialogResult(bool Canceled, object? Data)
{
    public static CbDialogResult Ok(object? data = null) => new(false, data);
    public static CbDialogResult Cancel() => new(true, null);
}

public sealed class CbDialogParameters : Dictionary<string, object?>
{
    public CbDialogParameters Add(string name, object? value) { this[name] = value; return this; }
}

public sealed record CbDialogOptions(
    CbDialogMaxWidth MaxWidth = CbDialogMaxWidth.Sm,
    bool FullWidth = true,
    bool CloseOnEscape = true,
    bool CloseOnScrim = true
);

public enum CbDialogMaxWidth { ExtraSmall, Sm, Md, Lg, Xl }

internal sealed record CbDialogRequest(
    Type DialogType,
    string Title,
    CbDialogParameters Parameters,
    CbDialogOptions Options,
    TaskCompletionSource<CbDialogResult> Tcs
);
```

Inside a TDialog (any ComponentBase), the dialog content reads its parameters from `[Parameter]` attributes (CbDialogHost sets them via reflection from the `CbDialogParameters` dictionary), and calls back to close via a CascadingParameter:

```csharp
[CascadingParameter] CbDialogInstance? DialogInstance { get; set; }
// ...
private void OnSave() => DialogInstance?.Close(CbDialogResult.Ok(_data));
```

Where `CbDialogInstance` is a small handle the host cascades into the rendered TDialog so the inner content can dismiss itself.

```csharp
using Microsoft.AspNetCore.Components;

namespace CookBot.Web.Services;

public sealed record CbDialogResult(bool Canceled, object? Data)
{
    public static CbDialogResult Ok(object? data = null) => new(false, data);
    public static CbDialogResult Cancel() => new(true, null);
}

public sealed class CbDialogParameters : Dictionary<string, object?>
{
    public new CbDialogParameters Add(string name, object? value) { this[name] = value; return this; }
}

public enum CbDialogMaxWidth { ExtraSmall, Sm, Md, Lg, Xl }

public sealed record CbDialogOptions(
    CbDialogMaxWidth MaxWidth = CbDialogMaxWidth.Sm,
    bool FullWidth = true,
    bool CloseOnEscape = true,
    bool CloseOnScrim = true)
{
    public static CbDialogOptions Default { get; } = new();
}

internal sealed record CbDialogRequest(
    Type DialogType,
    string Title,
    CbDialogParameters Parameters,
    CbDialogOptions Options,
    TaskCompletionSource<CbDialogResult> Tcs);

public interface ICbDialogService
{
    Task<CbDialogResult> ShowAsync<TDialog>(string title, CbDialogParameters? parameters = null, CbDialogOptions? options = null) where TDialog : ComponentBase;
    Task<CbDialogResult> ShowAsync(Type dialogType, string title, CbDialogParameters? parameters = null, CbDialogOptions? options = null);
    internal event Func<CbDialogRequest, Task>? OnRequest;
}

internal sealed class CbDialogService : ICbDialogService
{
    public event Func<CbDialogRequest, Task>? OnRequest;

    public Task<CbDialogResult> ShowAsync<TDialog>(string title, CbDialogParameters? parameters = null, CbDialogOptions? options = null) where TDialog : ComponentBase
        => ShowAsync(typeof(TDialog), title, parameters, options);

    public Task<CbDialogResult> ShowAsync(Type dialogType, string title, CbDialogParameters? parameters = null, CbDialogOptions? options = null)
    {
        var tcs = new TaskCompletionSource<CbDialogResult>(TaskCreationOptions.RunContinuationsAsynchronously);
        var req = new CbDialogRequest(
            dialogType,
            title,
            parameters ?? new CbDialogParameters(),
            options ?? CbDialogOptions.Default,
            tcs);

        if (OnRequest is null)
        {
            // No host mounted — fail fast so missing-host bugs surface in dev rather than hanging.
            tcs.SetException(new InvalidOperationException(
                "CbDialogHost is not mounted. Add <CbDialogHost /> to MainLayout.razor."));
        }
        else
        {
            // Fire-and-forget host invocation; host completes the Tcs when dialog closes.
            _ = OnRequest.Invoke(req);
        }
        return tcs.Task;
    }
}

public sealed class CbDialogInstance
{
    private readonly TaskCompletionSource<CbDialogResult> _tcs;
    public CbDialogInstance(TaskCompletionSource<CbDialogResult> tcs) { _tcs = tcs; }
    public void Close(CbDialogResult result) { if (!_tcs.Task.IsCompleted) _tcs.SetResult(result); }
    public void Cancel() => Close(CbDialogResult.Cancel());
}
```

**Create `src/CookBot.Web/wwwroot/js/cb-dialog.js`:**

```javascript
// CookBot dialog focus trap + body scroll lock + ESC handler.
// Phase 5 / DIALOG-01.
window.cookbotDialog = window.cookbotDialog || (function () {
  var traps = {};   // elementId -> { handleKey, previousFocus }
  var scrollLockCount = 0;

  function focusableSelector() {
    return 'a[href], area[href], button:not([disabled]), input:not([disabled]):not([type="hidden"]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])';
  }

  function trapFocus(elementId, dotnetRef) {
    var el = document.getElementById(elementId);
    if (!el) return;
    var prev = document.activeElement;
    var firstFocusable = el.querySelector(focusableSelector());
    if (firstFocusable) {
      try { firstFocusable.focus(); } catch (e) { /* ignore */ }
    }
    function handleKey(e) {
      if (e.key === 'Escape') {
        // Defer ESC to .NET — it will respect CloseOnEscape param
        if (dotnetRef && dotnetRef.invokeMethodAsync) {
          dotnetRef.invokeMethodAsync('OnEscape');
        }
        return;
      }
      if (e.key !== 'Tab') return;
      var nodes = el.querySelectorAll(focusableSelector());
      if (nodes.length === 0) { e.preventDefault(); return; }
      var first = nodes[0];
      var last = nodes[nodes.length - 1];
      if (e.shiftKey && document.activeElement === first) {
        last.focus(); e.preventDefault();
      } else if (!e.shiftKey && document.activeElement === last) {
        first.focus(); e.preventDefault();
      }
    }
    el.addEventListener('keydown', handleKey);
    traps[elementId] = { handleKey: handleKey, previousFocus: prev };

    // Body scroll lock
    if (scrollLockCount === 0) {
      document.body.style.overflow = 'hidden';
    }
    scrollLockCount++;
  }

  function releaseFocus(elementId) {
    var el = document.getElementById(elementId);
    var trap = traps[elementId];
    if (el && trap) { el.removeEventListener('keydown', trap.handleKey); }
    if (trap && trap.previousFocus && trap.previousFocus.focus) {
      try { trap.previousFocus.focus(); } catch (e) { /* ignore */ }
    }
    delete traps[elementId];

    scrollLockCount = Math.max(0, scrollLockCount - 1);
    if (scrollLockCount === 0) { document.body.style.overflow = ''; }
  }

  function bindOutsideClick(elementId, dotnetRef) {
    function handler(e) {
      var el = document.getElementById(elementId);
      if (!el) return;
      if (!el.contains(e.target)) {
        if (dotnetRef && dotnetRef.invokeMethodAsync) {
          dotnetRef.invokeMethodAsync('OnOutsideClick');
        }
      }
    }
    // Defer attaching by one tick to avoid catching the open-click event
    setTimeout(function () { document.addEventListener('mousedown', handler); }, 0);
    traps['outside_' + elementId] = { handleKey: handler };
  }

  function unbindOutsideClick(elementId) {
    var key = 'outside_' + elementId;
    var trap = traps[key];
    if (trap) {
      document.removeEventListener('mousedown', trap.handleKey);
      delete traps[key];
    }
  }

  return { trapFocus: trapFocus, releaseFocus: releaseFocus, bindOutsideClick: bindOutsideClick, unbindOutsideClick: unbindOutsideClick };
})();
```

**Create `src/CookBot.Web/Components/Dialogs/CbDialogHost.razor`** — the global mount that turns service requests into rendered components (D-24):

```razor
@namespace CookBot.Web.Components.Dialogs
@using CookBot.Web.Services
@using System.Reflection
@inject ICbDialogService DialogService

@foreach (var entry in _stack)
{
  <CbDialog @key="entry.Id" IsOpen="true"
            MaxWidth="entry.Options.MaxWidth"
            FullWidth="entry.Options.FullWidth"
            CloseOnEscape="entry.Options.CloseOnEscape"
            CloseOnScrim="entry.Options.CloseOnScrim"
            OnClose="@(() => HandleCancel(entry))">
    <Header>@entry.Title</Header>
    <Body>
      @* Render the dialog component dynamically; cascade an instance handle so it can self-close *@
      <CascadingValue Value="@(new CbDialogInstance(entry.Tcs))" IsFixed="true">
        <DynamicComponent Type="@entry.DialogType" Parameters="@entry.Parameters" />
      </CascadingValue>
    </Body>
  </CbDialog>
}

@code {
    private readonly List<HostEntry> _stack = new();

    private sealed record HostEntry(
        Guid Id,
        Type DialogType,
        string Title,
        Dictionary<string, object?> Parameters,
        CbDialogOptions Options,
        TaskCompletionSource<CbDialogResult> Tcs);

    protected override void OnInitialized()
    {
        DialogService.OnRequest += HandleRequest;
    }

    private Task HandleRequest(CbDialogRequest req)
    {
        var entry = new HostEntry(
            Guid.NewGuid(),
            req.DialogType,
            req.Title,
            new Dictionary<string, object?>(req.Parameters),
            req.Options,
            req.Tcs);
        _stack.Add(entry);

        // When the Tcs completes (dialog calls Close()), pop the entry.
        _ = entry.Tcs.Task.ContinueWith(_ =>
        {
            _stack.Remove(entry);
            InvokeAsync(StateHasChanged);
        }, TaskScheduler.Default);

        return InvokeAsync(StateHasChanged);
    }

    private void HandleCancel(HostEntry entry)
    {
        if (!entry.Tcs.Task.IsCompleted)
        {
            entry.Tcs.SetResult(CbDialogResult.Cancel());
        }
    }

    public void Dispose()
    {
        DialogService.OnRequest -= HandleRequest;
    }
}
```

`<DynamicComponent>` is Blazor's built-in component for rendering an arbitrary `Type`; it accepts `Parameters` as `IDictionary<string,object?>` and applies them via reflection. This sidesteps the need for a custom reflection-based parameter setter. The cascaded `CbDialogInstance` lets the inner dialog content call `DialogInstance.Close(result)` to dismiss.

Key correctness notes:
  - `OnRequest` is a multicast event; `CbDialogService.ShowAsync` wraps the invocation in `_ = OnRequest.Invoke(req)` so the host's StateHasChanged completes asynchronously without blocking the caller's await on `Tcs.Task`.
  - The `_stack` list supports D-23's "multiple-dialog stacking" requirement; each entry gets its own `<CbDialog>` instance with its own `_dialogId`.
  - The dialog's `OnClose` callback runs `HandleCancel` which sets the Tcs to `Cancel()` ONLY if not already completed — protects against the case where the inner dialog content already called `DialogInstance.Close(Ok(...))`.

```razor
@namespace CookBot.Web.Components.Dialogs
@using CookBot.Web.Services
@inject ICbToastService ToastService
@implements IDisposable

<div class="cb-toast-host">
  @foreach (var t in _toasts)
  {
    <div @key="t.Id" class="cb-toast @SeverityClass(t.Severity)">@t.Message</div>
  }
</div>

@code {
    private const int MaxStacked = 3;
    private const int LifetimeMs = 5000;
    private readonly List<CbToastMessage> _toasts = new();
    private readonly Dictionary<Guid, CancellationTokenSource> _timers = new();

    protected override void OnInitialized()
    {
        ToastService.OnToast += HandleToast;
    }

    private void HandleToast(CbToastMessage msg)
    {
        InvokeAsync(() =>
        {
            _toasts.Add(msg);
            while (_toasts.Count > MaxStacked) _toasts.RemoveAt(0);
            var cts = new CancellationTokenSource();
            _timers[msg.Id] = cts;
            _ = Task.Delay(LifetimeMs, cts.Token).ContinueWith(_ =>
            {
                if (cts.IsCancellationRequested) return;
                InvokeAsync(() =>
                {
                    _toasts.RemoveAll(x => x.Id == msg.Id);
                    _timers.Remove(msg.Id);
                    StateHasChanged();
                });
            }, TaskScheduler.Default);
            StateHasChanged();
        });
    }

    private static string SeverityClass(CbToastSeverity s) => s switch
    {
        CbToastSeverity.Success => "success",
        CbToastSeverity.Error   => "error",
        CbToastSeverity.Warning => "warning",
        _                        => "info"
    };

    public void Dispose()
    {
        ToastService.OnToast -= HandleToast;
        foreach (var cts in _timers.Values) cts.Cancel();
        _timers.Clear();
    }
}
```

<threat_model>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Server (existing) | CbDialogHost uses `DynamicComponent` to render a Type passed from server-side code. The Type is supplied by the *calling* server-side Razor code, NEVER by the browser. End-user input cannot select an arbitrary type to render. |
| JS → .NET via `[JSInvokable]` | `CbDialog.OnEscape()` and `CbDropdown.OnOutsideClick()` are exposed to the browser via `DotNetObjectReference`. These are pure UI hooks (close-on-escape, close-on-outside-click) — they take no parameters and only call internal close methods. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-04-01 | Elevation of Privilege via DynamicComponent.Type | CbDialogHost | mitigate | The `Type` is supplied by the server-side caller invoking `DialogService.ShowAsync<TDialog>(...)`. End-user input cannot reach `CbDialogHost`'s `req.DialogType` because the request is created inside server-side code that knows the concrete `TDialog`. The type-erased `ShowAsync(Type, ...)` overload is callable but only from server-side code. |
| T-05-04-02 | XSS via dialog parameters | CbDialogHost.DynamicComponent | accept | DynamicComponent applies parameters via reflection. Razor escapes by default at the component-render boundary. If a Phase 6/7 dialog binds a `[Parameter] public string SomeProp` that contains HTML and renders via `@SomeProp`, escape is automatic. If a future dialog uses `@((MarkupString)SomeProp)` with user input, that's a per-dialog vulnerability — out of scope for Phase 5. |
| T-05-04-03 | Denial of Service via toast spam | CbToastService | mitigate | `CbToastHost.MaxStacked = 3` hard-caps visible toasts. Old toasts are trimmed (`while (_toasts.Count > MaxStacked) _toasts.RemoveAt(0)`). A malicious / runaway caller emitting 10k toasts cannot OOM the browser — they're discarded as fast as they're added. The 5s auto-fade ensures stale toasts clear. |
| T-05-04-04 | Memory leak via dropdown / dialog `DotNetObjectReference` | CbDialog, CbDropdown | mitigate | Both implement `IDisposable` / `IAsyncDisposable` and call `_ref?.Dispose()`. Toast timers track CancellationTokenSource and cancel on Dispose. Standard Blazor circuit lifetime cleanup applies. |
| T-05-04-05 | Focus trap escape via injected DOM | CbDialog | accept | The trap uses `document.querySelector(focusableSelector())` which finds focusable nodes inside the dialog `<div>`. A user with DevTools can manually focus elements outside, but that's not a security-relevant attack — it's their own session. Real-attacker scenarios (browser exploits) are out of scope. |
| T-05-04-06 | Race in CbToastHost (concurrent OnToast invocations on Singleton service) | CbToastService → CbToastHost | mitigate | `CbToastHost.HandleToast` uses `InvokeAsync` to marshal onto the renderer's sync context. The `_toasts` list and `_timers` dictionary are touched only inside `InvokeAsync` callbacks. No lock needed because Blazor's renderer serializes the work. |
